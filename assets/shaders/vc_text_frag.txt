#version 330 core
                    in vec3 Normal;
                    in vec4 glPosition;
                    in vec3 glColor;
                    in vec2 TexC;


                    out vec4 FragColor;
                    uniform bool interpolation;
		    uniform sampler2D ourTexture;
                    uniform float opacity;
                    uniform vec3 viewPos;
                    uniform int numlight;
                    uniform vec3 tint;
                    uniform vec3 shaderColor;
struct Material{
vec3 ambient;
vec3 diffuse;
vec3 specular;
float shininess;
bool hastexture;
};
uniform Material material;


                    void main()
                    {
                    if (material.hastexture){
                    FragColor = texture(ourTexture, TexC)*vec4(tint,1.0);
}else{
                    if (interpolation) {
                    FragColor = vec4(glColor , opacity)*vec4(tint,1.0);
                    }
                    else{
                    FragColor=vec4(shaderColor, opacity)*vec4(tint,1.0);
                    }
}
                    }
